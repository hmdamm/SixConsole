local function RenderRosterGrid(targetSide, title)
    local indices = {}
    for i, op in ipairs(OPERATORS) do
        if op.side == targetSide then
            table.insert(indices, i)
        end
    end

    local cols = 5
    local colWidth = 14
    local total = #indices
    local rows = math.ceil(total / cols)

    OutputLogMessage("--- " .. title .. " ---\n")
    for r = 1, rows do
        local line = " "
        for c = 1, cols do
            local idxInTeam = ((r - 1) * cols) + c
            if idxInTeam <= total then
                local realIdx = indices[idxInTeam]
                local opName = OPERATORS[realIdx].name
                local prefix = (realIdx == currentOpIndex) and "[*]" or " . "
                local cell = prefix .. opName
                if #cell > (colWidth - 1) then cell = cell:sub(1, colWidth - 2) .. "." end
                cell = cell .. string.rep(" ", colWidth - #cell)
                line = line .. cell
            end
        end
        OutputLogMessage(line .. "\n")
    end
    OutputLogMessage("\n")
end

local function LogStatus()
    ClearLog()
    local activeOp = OPERATORS[currentOpIndex]

    OutputLogMessage("========================================================================\n")
    OutputLogMessage(" SCRIPT: [%s]  |  SENS MULTIPLIER: %.2fx  |  RF KEY: [%s]\n", isEnabled and "ACTIVE" or "DISABLED", recoilMultiplier, rfKey)
    OutputLogMessage("========================================================================\n\n")

    RenderRosterGrid("ATK", "ATTACKERS")
    RenderRosterGrid("DEF", "DEFENDERS")

    OutputLogMessage("========================================================================\n")
    OutputLogMessage(" SELECTED OPERATOR : [%s] %s\n", activeOp.side, activeOp.name)
    OutputLogMessage(" WEAPON LOADOUT (Total: %d) :\n", #activeOp.guns)

    for gIdx, gun in ipairs(activeOp.guns) do
        local effectivePull = Round(gun.pull * recoilMultiplier)
        local modeStr = gun.rapidFire and "DMR (RAPID FIRE)" or "FULL-AUTO RECOIL"
        local delayStr = gun.rapidFire and string.format("Hold %d-%dms | Gap %d-%dms", minHold, maxHold, minGap, maxGap) or string.format("%d-%d ms", minDelay, maxDelay)

        if gIdx == currentGunIndex then
            OutputLogMessage("  >>> [ACTIVATED]  (%d/%d) %-15s | %-16s | Pull: %3d px | %s <<<\n",
                gIdx, #activeOp.guns, gun.name, modeStr, effectivePull, delayStr)
        else
            OutputLogMessage("      [OPTION]     (%d/%d) %-15s | %-16s | Pull: %3d px | %s\n",
                gIdx, #activeOp.guns, gun.name, modeStr, effectivePull, delayStr)
        end
    end
    OutputLogMessage("========================================================================\n")
end

EnablePrimaryMouseButtonEvents(true)

function OnEvent(event, arg)

    if event == "MOUSE_BUTTON_PRESSED" and arg == 3 then
        isEnabled = not isEnabled
        LogStatus()
    end

    if event == "MOUSE_BUTTON_PRESSED" and arg == 5 then
        if IsKeyLockOn("capslock") then
            local currentOp = OPERATORS[currentOpIndex]
            currentGunIndex = (currentGunIndex % #currentOp.guns) + 1
        else
            currentOpIndex = (currentOpIndex % #OPERATORS) + 1
            currentGunIndex = 1
        end
        LogStatus()
    end

    if event == "MOUSE_BUTTON_PRESSED" and arg == 4 then
        if IsKeyLockOn("capslock") then
            local currentOp = OPERATORS[currentOpIndex]
            currentGunIndex = (currentGunIndex - 2 + #currentOp.guns) % #currentOp.guns + 1
        else
            currentOpIndex = (currentOpIndex - 2 + #OPERATORS) % #OPERATORS + 1
            currentGunIndex = 1
        end
        LogStatus()
    end

    if event == "MOUSE_BUTTON_PRESSED" and arg == 1 then
        if isEnabled then
            local op  = OPERATORS[currentOpIndex]
            local gun = op.guns[currentGunIndex]
            local effectivePull = Round(gun.pull * recoilMultiplier)

            -- MODE 1: Rapid Fire via Secondary Keybind ('P')
            if gun.rapidFire then
                repeat
                    PressKey(rfKey)
                    Sleep(math.random(minHold, maxHold))
                    ReleaseKey(rfKey)

                    if effectivePull > 0 then
                        local v = effectivePull + math.random(-vertVariance, vertVariance)
                        local h = math.random(-horizJitter, horizJitter)
                        MoveMouseRelative(h, v)
                    end

                    Sleep(math.random(minGap, maxGap))
                until not IsMouseButtonPressed(1)

            -- MODE 2: Full-Auto Recoil Control Only
            else
                repeat
                    if effectivePull > 0 then
                        local v = effectivePull + math.random(-vertVariance, vertVariance)
                        local h = math.random(-horizJitter, horizJitter)
                        MoveMouseRelative(h, v)
                    end
                    
                    Sleep(math.random(minDelay, maxDelay))
                until not IsMouseButtonPressed(1)
            end
        end
    end

end
